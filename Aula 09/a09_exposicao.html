Histórico com rastreabilidade

SELECT m.data, p.nome, m.tipo, u.nome AS responsavel
FROM movimentacao m
JOIN produto p ON m.id_produto=p.id_produto
JOIN usuario u ON m.id_usuario=u.id_usuario;
